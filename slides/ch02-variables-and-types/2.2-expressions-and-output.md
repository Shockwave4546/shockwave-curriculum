      <ul class="bullets">
        <li><span class="bul">▸</span><span><span class="me">System.out.println()</span> prints and moves to a new line; <span class="me">print()</span> doesn't.</span></li>
        <li><span class="bul">▸</span><span><span class="k">int / int</span> truncates the decimal — make one operand a <span class="k">double</span> to keep it.</span></li>
        <li><span class="bul">▸</span><span><span class="k">%</span> (remainder) is the leftover after integer division — great for "every Nth loop."</span></li>
      </ul>
    </div>

      <span class="cl"><span class="k">int</span> ticks = <span class="n">7</span>, cycle = <span class="n">3</span>;</span>
      <span class="cl">System.out.<span class="me">println</span>(ticks / cycle); <span class="c">// 2 — truncated</span></span>
      <span class="cl">System.out.<span class="me">println</span>(ticks % cycle); <span class="c">// 1 — remainder</span></span>
      <span class="cl">&nbsp;</span>
      <span class="cl"><span class="k">double</span> avgSpeed = (<span class="k">double</span>) ticks / cycle;</span>
